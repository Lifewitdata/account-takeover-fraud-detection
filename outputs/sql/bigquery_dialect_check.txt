one row per login attempt, 15 past-only features */ /* ---------------------------------------------------------------------------- */ CREATE OR REPLACE VIEW "project"."dataset"."login_features" AS WITH base AS (SELECT *, TO_TIMESTAMP(600 * CAST(EPOCH(timestamp) AS BIGINT) // 600) AS t10 /* 10-minute bucket for velocity features */ FROM "project"."dataset"."login_attempts"), with_prev AS (SELECT b.*, LAG(timestamp) OVER w AS prev_ts, LAG(ip_lat) OVER w AS prev_lat, LAG(ip_lon) OVER w AS prev_lon, LAG(device_id) OVER w AS prev_device, LAG(ip_country) OVER w AS prev_country FROM base AS b WINDOW w AS (PARTITION BY user_id ORDER BY timestamp NULLS FIRST)), movement AS (SELECT *, 6371 /* haversine distance (km) between consecutive logins of the same user */ * 2 * ASIN(SQRT(POWER(SIN((RADIANS(ip_lat) - RADIANS(prev_lat)) / 2), 2) + COS(RADIANS(prev_lat)) * COS(RADIANS(ip_lat)) * POWER(SIN((RADIANS(ip_lon) - RADIANS(prev_lon)) / 2), 2))) AS distance_km, DATE_DIFF('MINUTE', prev_ts, timestamp) / 60.0 AS hours_since_last FROM with_prev), flags AS (SELECT *, COALESCE(distance_km / NULLIF(hours_since_last, 0), 0) AS speed_kmh, CASE WHEN COALESCE(distance_km, 0) > 500 AND COALESCE(distance_km / NULLIF(hours_since_last, 0), 0) > 800 THEN 1 ELSE 0 END AS impossible_travel, CASE WHEN NOT prev_device IS NULL AND device_id <> prev_device THEN 1 ELSE 0 END AS new_device, CASE WHEN NOT prev_country IS NULL AND ip_country <> prev_country THEN 1 ELSE 0 END AS new_country, EXTRACT(HOUR FROM timestamp) AS hour, EXTRACT(DAYOFWEEK FROM timestamp) AS dow /* 1 = Sunday ? 7 = Saturday */ FROM movement), user_vel AS (SELECT user_id, t10, COUNT(*) AS u_att, COUNT_IF(login_result = 'failed') AS u_failed FROM base GROUP BY 1, 2), ip_vel AS (SELECT ip_address, t10, COUNT(*) AS i_att, COUNT_IF(login_result = 'failed') AS i_failed, COUNT(DISTINCT user_id) AS i_users FROM base GROUP BY 1, 2), device_sharing AS (SELECT device_id, COUNT(DISTINCT user_id) AS device_n_users FROM base GROUP BY 1), ip_sharing AS (SELECT ip_address, COUNT(DISTINCT user_id) AS ip_n_users FROM base GROUP BY 1) SELECT f.* EXCLUDE (t10, prev_ts, prev_lat, prev_lon, prev_device, prev_country), uv.u_att, uv.u_failed, iv.i_att, iv.i_failed, iv.i_users, ds.device_n_users, ish.ip_n_users FROM flags AS f LEFT JOIN user_vel AS uv USING (user_id, t10) LEFT JOIN ip_vel AS iv USING (ip_address, t10) LEFT JOIN device_sharing AS ds USING (device_id) LEFT JOIN ip_sharing AS ish USING (ip_address);

   Interpretation: distance_km differs everywhere because the notebook's
   haversine_km scales the longitude term by cos(lat2) ** 2 while the BigQuery
   file writes the textbook cos(lat1) * cos(lat2). hours_since_last / speed_kmh /
   impossible_travel differ because BigQuery's TIMESTAMP_DIFF(..., MINUTE) floors
   the gap to whole minutes. dow differs on every row: DAYOFWEEK is Sunday = 1..7,
   pandas dayofweek is Monday = 0..6 (sqlglot renders it as DuckDB dayofweek,
   Sunday = 0..6, a third scale). new_device / new_country differ on 6 rows only,
   which is the unqualified LAG ORDER BY timestamp over 36 tied (user_id,
   timestamp) pairs - the original has no deterministic tie-break, so those flags
   are not reproducible. u_att / u_failed / i_att / i_failed / i_users differ
   because sqlglot renders BigQuery's 600 * DIV(UNIX_SECONDS(ts), 600) as
   600 * CAST(EPOCH(ts) AS BIGINT) // 600, and in DuckDB * and // bind left to
   right, so the multiplication applies before the floor division and the
   10-minute bucket collapses to per-second. Transpiled text needs review, not
   blind execution; sql/duckdb/login_features.sql is the version that is proven.
